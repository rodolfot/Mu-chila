CREATE PROCEDURE [dbo].[WZ_HuntingRecordInfoUserOpenLoad]
	@AccountID	VARCHAR(10)
	,@Name		VARCHAR(10)
AS  
SET NOCOUNT ON
SET TRANSACTION ISOLATION LEVEL READ UNCOMMITTED

SELECT UserOpen
FROM dbo.T_HuntingRecordOption
WHERE AccountID	= @AccountID	
	AND	Name = @Name


GO
