CREATE PROCEDURE [dbo].[WZ_HuntingRecordDelete]
	@AccountID	VARCHAR(10)
	,@Name		VARCHAR(10)
	,@MapIndex	INT
	,@mYear		INT
	,@mMonth	INT
	,@mDay		INT
AS  
SET NOCOUNT ON
SET TRANSACTION ISOLATION LEVEL READ UNCOMMITTED

DELETE FROM dbo.T_HuntingRecord 
WHERE AccountID		= @AccountID	
	AND	Name		= @Name		
	AND	MapIndex	= @MapIndex
	AND	mYear		= @mYear
	AND	mMonth		= @mMonth
	AND	mDay		= @mDay


GO
